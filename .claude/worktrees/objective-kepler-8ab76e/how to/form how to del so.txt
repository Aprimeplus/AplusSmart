-- ขั้นตอนที่ 2: ลบข้อมูลลูกที่อ้างอิงถึง PO ของ SO นี้ก่อน
DELETE FROM purchase_order_items
WHERE purchase_order_id IN (SELECT id FROM purchase_orders WHERE so_number = 'SO809ON002');

DELETE FROM purchase_order_payments
WHERE purchase_order_id IN (SELECT id FROM purchase_orders WHERE so_number = 'SO809ON002');

-- ขั้นตอนที่ 3: ลบข้อมูลแม่ (PO และ SO)
DELETE FROM purchase_orders
WHERE so_number = 'SO809ON002';

DELETE FROM commissions
WHERE so_number = 'SO809ON002';

-- ขั้นตอนที่ 4: ตรวจสอบผลลัพธ์ (แนะนำให้ทำก่อน COMMIT)
-- ถ้า query ทั้งสองนี้ไม่แสดงผลลัพธ์ แสดงว่าข้อมูลถูกลบเรียบร้อยแล้ว
SELECT so_number FROM commissions WHERE so_number = 'SO809ON002';
SELECT so_number FROM purchase_orders WHERE so_number = 'SO809ON002';

-- ขั้นตอนที่ 5: ยืนยันการลบ หรือ ยกเลิก
-- หากตรวจสอบแล้วถูกต้อง ให้รันคำสั่งนี้เพื่อยืนยันการลบทั้งหมด
COMMIT;
